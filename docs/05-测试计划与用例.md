```java
@ExtendWith(MockitoExtension.class)
class SessionCoordinatorTest {

    @Mock SessionRepository sessionRepository;
    @Mock OperationRepository operationRepository;
    @Mock NodeCommandGateway nodeCommandGateway;
    @Mock OutboxPublisher outboxPublisher;

    @InjectMocks SessionCoordinator coordinator;

    @Test
    void shouldRejectSecondActiveOperation() {
        // Given
        operationRepository.insert(activeOperation("session-1"));

        // When & Then
        assertThrows(
            ActiveOperationExistsException.class,
            () -> coordinator.handle(
                new StartSession("session-1", "runtime-1", "idem-2")
            )
        );
    }

    @Test
    void shouldCreateStartOperation() {
        // Given
        var session = createSession("session-1", SessionState.CREATED);
        when(sessionRepository.require("session-1")).thenReturn(session);
        when(operationRepository.ensureNoActiveOperation("session-1"))
            .thenReturn(Optional.empty());

        // When
        var result = coordinator.handle(
            new StartSession("session-1", "runtime-1", "idem-1")
        );

        // Then
        assertThat(result.status()).isEqualTo(CoordinatorResult.Status.ACCEPTED);
        verify(operationRepository).insert(any(ExclusiveOperation.class));
        verify(nodeCommandGateway).send(any(NodeCommand.class));
    }

    @Test
    void shouldHandleNodeEventWithCorrectEpoch() {
        // Given
        var session = createSession("session-1", SessionState.STARTING);
        var operation = createActiveOperation("session-1");
        when(sessionRepository.require("session-1")).thenReturn(session);
        when(operationRepository.findActive("session-1"))
            .thenReturn(Optional.of(operation));

        var event = new RuntimeStartedEvent("session-1", 12345L, 1L, "http://localhost:9222");

        // When
        var result = coordinator.handle(
            new NodeEventReceived("session-1", event)
        );

        // Then
        assertThat(result.status()).isEqualTo(CoordinatorResult.Status.COMPLETED);
        verify(sessionRepository).updateWithExpectedEpoch(any(), eq(0L));
    }

    @Test
    void shouldRejectStaleNodeEvent() {
        // Given
        var session = createSession("session-1", SessionState.RUNNING);
        session = session.nextContextEpoch("node-2", "build-2", 2L);
        when(sessionRepository.require("session-1")).thenReturn(session);

        var event = new RuntimeStartedEvent("session-1", 12345L, 1L, "http://localhost:9222");

        // When
        var result = coordinator.handle(
            new NodeEventReceived("session-1", event)
        );

        // Then
        assertThat(result.status()).isEqualTo(CoordinatorResult.Status.REJECTED);
        assertThat(result.reason()).isEqualTo("STALE_CONTEXT_EPOCH");
    }

    @Test
    void shouldHandleTimeoutByAbortingOperation() {
        // Given
        var operation = createActiveOperation("session-1");
        when(operationRepository.findActive("session-1"))
            .thenReturn(Optional.of(operation));

        // When
        var result = coordinator.handle(
            new OperationTimedOut("session-1", operation.operationId())
        );

        // Then
        assertThat(result.status()).isEqualTo(CoordinatorResult.Status.COMPLETED);
        verify(operationRepository).transition(
            eq(operation.operationId()),
            eq(OperationState.ACTIVE),
            eq(OperationState.TIMED_OUT)
        );
    }
}
```

```typescript
import { test, expect } from '@playwright/test';

test.describe('Session Lifecycle', () => {
  test('should complete full session lifecycle', async ({ page }) => {
    // Create session
    const createResponse = await page.request.post('/api/v1/sessions', {
      data: {
        tenantId: 'test-tenant',
        profileId: 'test-profile',
      },
      headers: { 'Idempotency-Key': `idem-${Date.now()}` },
    });
    const { sessionId } = await createResponse.json();

    // Start session
    const startResponse = await page.request.post(
      `/api/v1/sessions/${sessionId}:start`
    );
    expect(startResponse.status()).toBe(202);

    // Wait for running state
    await expect(async () => {
      const response = await page.request.get(
        `/api/v1/sessions/${sessionId}`
      );
      const session = await response.json();
      expect(session.state).toBe('RUNNING');
    }).toPass({ timeout: 30000 });

    // Get session details
    const sessionResponse = await page.request.get(
      `/api/v1/sessions/${sessionId}`
    );
    const session = await sessionResponse.json();
    expect(session.browserGeneration).toBeGreaterThan(0);
    expect(session.nodeId).toBeTruthy();

    // Terminate session
    const terminateResponse = await page.request.post(
      `/api/v1/sessions/${sessionId}:terminate`
    );
    expect(terminateResponse.status()).toBe(202);

    // Wait for terminated state
    await expect(async () => {
      const response = await page.request.get(
        `/api/v1/sessions/${sessionId}`
      );
      const session = await response.json();
      expect(session.state).toBe('TERMINATED');
    }).toPass({ timeout: 30000 });
  });

  test('should handle duplicate start requests', async ({ page }) => {
    const idempotencyKey = `idem-${Date.now()}`;

    // Create session
    const createResponse = await page.request.post('/api/v1/sessions', {
      data: { tenantId: 'test-tenant', profileId: 'test-profile' },
      headers: { 'Idempotency-Key': idempotencyKey },
    });
    const { sessionId } = await createResponse.json();

    // Start twice with same idempotency key
    const start1 = await page.request.post(
      `/api/v1/sessions/${sessionId}:start`
    );
    const start2 = await page.request.post(
      `/api/v1/sessions/${sessionId}:start`
    );

    // Both should succeed with same operation
    const op1 = await start1.json();
    const op2 = await start2.json();
    expect(op1.operationId).toBe(op2.operationId);
  });
});
```
